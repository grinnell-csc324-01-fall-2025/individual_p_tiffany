// 预设 prompt 和对应回应 const responses = { "😟 I'm feeling a bit down today.": "I'm sorry to hear that. Can you tell me what’s been on your mind lately? What do you think might help you feel a little better right now?", "😊 I had a really good day!": "That’s wonderful! 🎉 What made today feel so good for you? Do you think it’s something you could bring into tomorrow too?", "😔 I feel lonely sometimes.": "Feeling lonely can be tough. 💙 Who do you usually reach out to when you feel this way? Would you like to think about new ways to connect with others?", "😡 I'm frustrated about work.": "Work stress can really drain your energy. What part of work is bothering you the most? Have you tried talking to someone about it?", "😴 I’m just really tired.": "Sounds like your body is asking for rest. 😴 Have you been sleeping well lately? What’s one small thing you could do tonight to recharge?" }; // 当点击 prompt 时，显示不同的回应 function sendPrompt(promptText) { const chatBox = document.getElementById("chat-box"); // 显示用户输入 const userMsg = document.createElement("div"); userMsg.className = "chat user"; userMsg.textContent = promptText; chatBox.appendChild(userMsg); // 显示 chatbot 回应 const botMsg = document.createElement("div"); botMsg.className = "chat bot"; botMsg.textContent = responses[promptText] || "Thanks for sharing!"; chatBox.appendChild(botMsg); chatBox.scrollTop = chatBox.scrollHeight; // 自动滚动 }
